fn address(id: &str, role: Option<SpiritRole>) -> FrameAddress {
    FrameAddress {
        spirit_id: SpiritId(id.into()),
        host_id: Some(HostId("host-17-3a".into())),
        role,
    }
}

fn probe_frame_round_trip(frame: &IacFrame) -> Result<(), String> {
    let original = maos_wasm_host::codec::encode_cbor(frame).map_err(|e| e.to_string())?;
    let extended = bridge::lift(bridge::lower(frame).map_err(|e| e.to_string())?)
        .map_err(|e| e.to_string())?;
    let extended_bytes = maos_wasm_host::codec::encode_cbor(&extended).map_err(|e| e.to_string())?;
    if original != extended_bytes {
        return Err("extended bridge changed canonical CBOR".into());
    }

    let head = maos_wasm_host::frame_bridge::lift(maos_wasm_host::frame_bridge::lower(frame))
        .map_err(|e| e.to_string())?;
    let head_bytes = maos_wasm_host::codec::encode_cbor(&head).map_err(|e| e.to_string())?;
    if original == head_bytes {
        return Err("HEAD control unexpectedly preserved canonical CBOR".into());
    }
    let FramePayload::TaskAssign(head_task) = &head.payload else {
        return Err("HEAD control changed the task-assignment payload variant".into());
    };
    if head.intent != IntentClass::Readonly
        || head.consent_envelope.is_some()
        || !head.intent_lineage.is_empty()
        || !head_task.scope.is_empty()
    {
        return Err("HEAD control did not exhibit its documented field loss".into());
    }

    println!("extended_cbor_equal=true bytes={}", original.len());
    println!("head_control_cbor_equal=false head_bytes={}", head_bytes.len());
    println!("head_control_defaults=readonly/none/empty-lineage/empty-scope");
    Ok(())
}

fn probe_frame_growth_skew() -> Result<(), String> {
    let mut config = Config::new();
    config.wasm_component_model(true);
    let engine = Engine::new(&config).map_err(|e| e.to_string())?;
    let component = Component::from_file(
        &engine,
        "tests/fixtures/wasm/echo_spirit_component.wasm",
    )
    .map_err(|e| e.to_string())?;
    let mut store = Store::new(&engine, maos_wasm_host::host_state::HostState::new());
    let mut linker = Linker::new(&engine);
    wasmtime_wasi::p2::add_to_linker_sync(&mut linker).map_err(|e| e.to_string())?;
    match bindings::Spirit::instantiate(&mut store, &component, &linker) {
        Ok(_) => Err("old @1.0.0 fixture instantiated against frame-grown @1.1.0 WIT".into()),
        Err(error) => {
            let message = format!("{error:?}");
            if !message.contains("expected record of 11 fields, found 8 fields") {
                return Err(format!(
                    "expected frame-record type mismatch after satisfying WASI imports; got: {message}"
                ));
            }
            println!("frame_growth_skew=type-mismatch");
            println!("frame_growth_error={message}");
            Ok(())
        }
    }
}

fn main() -> Result<(), String> {
    let frame = populated_frame();
    probe_frame_round_trip(&frame)?;
    probe_frame_growth_skew()?;
    Ok(())
}
