# Story 17-3d — throwaway CI route (D-17-3d-FW-10). Lives ONLY on
# `ci/17-3d-*` branches; never lands on main. Read-only permissions, no secrets.
# Jobs copy the shipped `discipline.yml` steps verbatim: exit lines 2 and 2b
# (`epic-17-exit`), the 17-3d recall gate (`wasm-host-tests`), and the migrated
# TypeScript driver (`example-spirit-ts-tests`), without unrelated `needs`.
name: ci-17-3d-verify
on:
  push:
    branches: ['ci/17-3d-*']
permissions:
  contents: read
jobs:
  exit-line-2-2b:
    strategy:
      fail-fast: false
      matrix:
        image: [ubuntu-24.04, ubuntu-26.04]
    runs-on: ${{ matrix.image }}
    timeout-minutes: 120
    steps:
      - uses: actions/checkout@v5
      - uses: actions/setup-node@v5
        with:
          node-version: '24'
      - uses: dtolnay/rust-toolchain@v1
        with:
          toolchain: stable
      - uses: Swatinem/rust-cache@v2
        with:
          key: ci-17-3d-exit-${{ matrix.image }}-${{ hashFiles('**/Cargo.lock') }}
      - name: Keep build and image temporaries off /tmp
        run: echo "TMPDIR=$RUNNER_TEMP" >> "$GITHUB_ENV"
      - name: Provenance
        run: |
          echo "::notice title=17-3d provenance (${{ matrix.image }})::sha=${GITHUB_SHA} tree=$(git rev-parse HEAD^{tree}) rustc=$(rustc --version | tr ' ' '_')"
      - name: Componentize the TypeScript Spirit (line 2 prerequisite)
        working-directory: examples/example-spirit-ts
        run: npm ci && npm run build && npm run componentize
      - name: Epic-17 exit line 2 — the TypeScript Spirit launch (Story 17-3c)
        shell: bash
        run: |
          set +e
          cargo build --release -p maos-bin -p maos-wasm-host --features maos-bin/wasm-host --bin maos --bin maos-wasm-runner &&
            sudo systemd-run --unit=maos-17-3c-proof --wait --pipe --collect \
              --uid="$(id -u)" --gid="$(id -g)" \
              --property=Delegate=yes --property=CPUAccounting=yes --property=MemoryAccounting=yes \
              --property="WorkingDirectory=$PWD" \
              "$PWD/scripts/run_17_3c_delegated.sh" target/release/maos run examples/example-spirit-ts/manifest.toml --once 2>&1 | tee line2.log
          rc=${PIPESTATUS[0]}
          if grep -q '^SKIP ' line2.log; then rc=1; echo "::error title=Epic-17 exit line 2 (${{ matrix.image }})::a SKIP line was printed"; fi
          if [ "$rc" -ne 0 ]; then
            echo "::error title=Epic-17 exit line 2 (${{ matrix.image }})::rc=${rc} $(tail -n 3 line2.log | paste -sd' ' -)"
          else
            echo "::notice title=Epic-17 exit line 2 (${{ matrix.image }})::rc=0"
          fi
          exit "$rc"
      - name: Epic-17 exit line 2b — the TypeScript Spirit's recall rows (Story 17-3d)
        shell: bash
        run: |
          set +e
          env -u MAOS_HOME -u MAOS_AUDIT_DB -u XDG_DATA_HOME HOME="$(getent passwd "$(id -u)" | cut -d: -f6)" \
            target/release/maos audit query --spirit example-spirit-ts --format plain > line2b.log 2> line2b.err
          rc=$?
          cat line2b.log line2b.err
          if [ "$rc" -ne 0 ]; then
            echo "::error title=Epic-17 exit line 2b (${{ matrix.image }})::maos audit query exited ${rc}: $(tail -n 2 line2b.err | paste -sd' ' -)"
            exit "$rc"
          fi
          python3 scripts/epic_17_exit_line2b.py line2b.log | tee line2b.verdict
          rc=${PIPESTATUS[0]}
          while IFS= read -r line; do
            if [ "$rc" -eq 0 ]; then echo "::notice title=Epic-17 exit line 2b (${{ matrix.image }})::${line}"
            else echo "::error title=Epic-17 exit line 2b (${{ matrix.image }})::${line}"; fi
          done < line2b.verdict
          exit "$rc"
      - uses: actions/upload-artifact@v4
        if: always()
        with:
          name: ci-17-3d-exit-${{ matrix.image }}
          path: |
            line2.log
            line2b.log
            line2b.err
          if-no-files-found: warn

  recall-17-3d:
    strategy:
      fail-fast: false
      matrix:
        image: [ubuntu-24.04, ubuntu-26.04, ubuntu-24.04-arm, ubuntu-26.04-arm]
    runs-on: ${{ matrix.image }}
    timeout-minutes: 90
    steps:
      - uses: actions/checkout@v5
      - uses: dtolnay/rust-toolchain@v1
        with:
          toolchain: stable
          targets: wasm32-wasip2
      - uses: Swatinem/rust-cache@v2
        with:
          key: ci-17-3d-recall-${{ matrix.image }}-${{ hashFiles('**/Cargo.lock') }}
      - name: Fetch locked standalone fixture dependencies for offline process gates
        run: |
          for fixture in \
            crates/maos-wasm-host/guests/echo-spirit \
            crates/maos-wasm-host/guests/routed-relay \
            crates/maos-wasm-host/guests/recall-probe \
            crates/maos-wasm-host/guests/compat-echo-2-0 \
            crates/maos-wasm-host/guests/equiv-fixture/native-twin; do
            cargo fetch --locked --manifest-path "$fixture/Cargo.toml"
          done
          cargo fetch --locked
      - name: maos-wasm-host recall runner + wire suites (Story 17-3d)
        shell: bash
        run: |
          set +e
          cargo test --locked -p maos-wasm-host --lib recall_wire 2>&1 | tee wire.log
          rc1=${PIPESTATUS[0]}
          cargo test --locked -p maos-wasm-host --test recall_runner_17_3d --test wit_corpus --test t2_sandbox_kill -- --nocapture --test-threads=1 2>&1 | tee runner.log
          rc2=${PIPESTATUS[0]}
          rc=$(( rc1 || rc2 ))
          if grep -q '^SKIP ' runner.log; then rc=1; echo "::error title=17-3d runner (${{ matrix.image }})::a SKIP line was printed"; fi
          echo "::notice title=17-3d runner (${{ matrix.image }})::rc=${rc} $(grep '^test result' wire.log runner.log | paste -sd' ' -)"
          exit "$rc"
      - name: In-band recall through the admitted session (Story 17-3d)
        shell: bash
        run: |
          set +e
          set -o pipefail
          python3 scripts/run_cgroup_ceiling_17_2.py run --mode delegated --unit maos-17-2-wasm-recall --timeout 2400 -- \
            bash -euo pipefail -c 'cargo test --locked -p maos-bin --features network,wasm-host --test recall_17_3d -- --nocapture --test-threads=1 2>&1 | tee recall-17-3d.log; test ${PIPESTATUS[0]} -eq 0'
          rc=$?
          fail=0
          [ "$rc" -eq 0 ] || { echo "::error title=17-3d recall (${{ matrix.image }})::cargo test rc=$rc"; fail=1; }
          grep -q '^test result: ok\. 16 passed;' recall-17-3d.log || { echo "::error title=17-3d recall (${{ matrix.image }})::missing 'test result: ok. 16 passed;'"; fail=1; }
          required=()
          for name in checked_in_researcher_admits_both_scopes_and_native_port_rows_pass_fr4 \
                      configured_sso_preserves_r1_r2_r3_order_and_never_recalls_identity_asserted \
                      consent_past_and_future_withheld_by_emitter_scope_no_consent_evaluation_exists_yet \
                      cross_boot_same_pid_is_not_recalled_and_fetch_is_scope_violation \
                      empty_page_has_one_token_bearing_call_and_no_receipt \
                      exhausted_quota_has_receipt_without_tokenless_fallback_call \
                      junk_after_tag_closes_once_with_typed_protocol_end \
                      operator_deny_overrides_manifest_and_has_receipt_only \
                      oversized_fetch_reply_is_typed_and_next_call_same_turn_still_works \
                      own_frame_second_turn_recalls_kernel_minted_id_kind_intent_and_fetches_stored_payload \
                      own_receipt_and_authorization_ids_are_never_recalled_and_fetch_is_scope_violation \
                      seed_fetch_r1_r2_r3_precede_authorization_and_fr4_succeeds \
                      sixty_fifth_recall_closes_channel_once_with_typed_protocol_end \
                      tagged_record_between_exports_closes_once_with_typed_protocol_end \
                      ungranted_manifest_denies_with_receipt_only \
                      unknown_tag_closes_once_with_typed_protocol_end; do
            required+=(--required-test "$name")
          done
          if grep -q '^SKIP ' recall-17-3d.log; then
            grep '^SKIP ' recall-17-3d.log | while IFS= read -r line; do
              echo "::error title=17-3d recall (${{ matrix.image }})::${line}"
            done
            fail=1
          fi
          [ "$fail" -eq 0 ] || grep -E 'panicked at' recall-17-3d.log | head -5 | while IFS= read -r line; do
            echo "::error title=17-3d recall (${{ matrix.image }})::${line}"
          done
          echo "::notice title=17-3d recall (${{ matrix.image }})::rc=$rc fail=$fail $(grep '^test result' recall-17-3d.log)"
          python3 scripts/run_cgroup_ceiling_17_2.py check-log --exit-code "$rc" --expected-count 16 "${required[@]}" recall-17-3d.log || fail=1
          exit "$fail"
      - name: Admitted 17-3c gates still green (frames unchanged)
        shell: bash
        run: |
          set +e
          set -o pipefail
          python3 scripts/run_cgroup_ceiling_17_2.py run --mode delegated --unit maos-17-2-wasm-host --timeout 2400 -- \
            bash -euo pipefail -c 'cargo test --locked -p maos-bin --features network,wasm-host --test spawned_scope_17_3c -- --nocapture --test-threads=1 2>&1 | tee spawned-17-3c.log; test ${PIPESTATUS[0]} -eq 0'
          rc=$?
          echo "::notice title=17-3c gates under 17-3d (${{ matrix.image }})::rc=$rc $(grep '^test result' spawned-17-3c.log)"
          python3 scripts/run_cgroup_ceiling_17_2.py check-log --exit-code "$rc" --expected-count 22 spawned-17-3c.log || rc=1
          exit "$rc"
      - uses: actions/upload-artifact@v4
        if: always()
        with:
          name: ci-17-3d-recall-${{ matrix.image }}
          path: |
            wire.log
            runner.log
            recall-17-3d.log
            spawned-17-3c.log
          if-no-files-found: warn

  ts-component:
    runs-on: ubuntu-latest
    timeout-minutes: 45
    steps:
      - uses: actions/checkout@v5
      - uses: actions/setup-node@v5
        with:
          node-version: '24'
      - uses: dtolnay/rust-toolchain@v1
        with:
          toolchain: stable
      - uses: Swatinem/rust-cache@v2
        with:
          key: ci-17-3d-ts-${{ hashFiles('**/Cargo.lock') }}
      - name: Build, test, and componentize the TypeScript scaffold
        shell: bash
        run: |
          set +e
          ( cd examples/example-spirit-ts && npm ci && npm test && npm run build && npm run componentize ) 2>&1 | tee example-spirit-ts.log
          rc=${PIPESTATUS[0]}
          echo "::notice title=17-3d TS scaffold::rc=${rc} $(grep -E '^[[:space:]]*Tests[[:space:]]+' example-spirit-ts.log | paste -sd';' -)"
          exit "$rc"
      - name: Drive the TypeScript component through the release runner (migrated ts_component_17_3b)
        shell: bash
        run: |
          set +e
          cargo build --locked --release -p maos-wasm-host --bin maos-wasm-runner &&
            MAOS_TS_COMPONENT=$PWD/examples/example-spirit-ts/dist/spirit.wasm cargo test --locked --release -p maos-wasm-host --test ts_component_17_3b -- --ignored --nocapture 2>&1 | tee ts-component-driver.log
          rc=${PIPESTATUS[0]}
          echo "::notice title=17-3d TS driver::rc=${rc} $(grep '^test result' ts-component-driver.log)"
          exit "$rc"
