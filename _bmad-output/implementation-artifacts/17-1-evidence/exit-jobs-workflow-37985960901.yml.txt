name: 17-1 epic exit and unmasked T3 jobs
on:
  push:
    branches: [ci/17-1-exit-20261009t2016]
permissions:
  contents: read
jobs:
  epic-17-exit:
    strategy:
      fail-fast: false
      matrix:
        image:
        - ubuntu-24.04
        - ubuntu-26.04
    runs-on: ${{ matrix.image }}
    timeout-minutes: 60
    steps:
    - uses: actions/checkout@v5
    - uses: actions/setup-node@v5
      with:
        node-version: '24'
    - uses: dtolnay/rust-toolchain@v1
      with:
        toolchain: stable
    - uses: Swatinem/rust-cache@v2
      with:
        key: epic-17-exit-${{ matrix.image }}-${{ hashFiles('**/Cargo.lock') }}
    - name: Keep build and image temporaries off /tmp
      run: echo "TMPDIR=$RUNNER_TEMP" >> "$GITHUB_ENV"
    - uses: ./.github/actions/provision-t3-test-substrate
    - name: Build the Worker root and the static probe fixture
      run: 'cargo build --locked -p maos-bin --bin maos

        cargo build --locked -p maos-cli --bin maosctl

        RUSTFLAGS=''-C target-feature=+crt-static'' cargo build --locked -p worker --bin worker-probe-fixture --target x86_64-unknown-linux-gnu

        '
    - name: "Epic-17 exit lines 1 and 1b \u2014 Worker sandbox, egress refusal, audit query"
      run: python3 scripts/epic_17_exit_line1.py --maos target/debug/maos --probe target/x86_64-unknown-linux-gnu/debug/worker-probe-fixture --output epic-17-exit-line1 --image ${{ matrix.image }}
    - name: Build the forbidden-syscall probes (line 3 prerequisite)
      run: 'cargo build --locked --release --manifest-path crates/maos-wasm-host/test-fixtures/forbidden-syscall-probe/Cargo.toml

        cargo build --locked --release --manifest-path crates/maos-escape-detector/test-fixtures/forbidden-syscall-probe/Cargo.toml

        '
    - name: "Epic-17 exit line 3 \u2014 T2 kill (Story 17-6)"
      shell: bash
      run: "set +e\ncargo test --locked -p maos-wasm-host --test t2_sandbox_kill -- --nocapture 2>&1 | tee t2.log\nrc=${PIPESTATUS[0]}\ngrep -E 'panicked at|rule 11' t2.log | head -3 | while IFS= read -r line; do\n  echo \"::error title=Epic-17 exit line 3 (${{ matrix.image }})::${line}\"\ndone\nif grep -q '^SKIP ' t2.log; then rc=1; echo \"::error title=Epic-17 exit line 3 (${{ matrix.image }})::a SKIP line was printed\"; fi\necho \"::notice title=Epic-17 exit line 3 (${{ matrix.image }})::rc=${rc} $(grep '^test result' t2.log)\"\nexit \"$rc\"\n"
    - name: Componentize the TypeScript Spirit (line 2 prerequisite)
      working-directory: examples/example-spirit-ts
      run: npm ci && npm run build && npm run componentize
    - name: "Epic-17 exit line 2 \u2014 the TypeScript Spirit launch (Story 17-3c)"
      shell: bash
      run: "set +e\ncargo build --release -p maos-bin -p maos-wasm-host --features maos-bin/wasm-host --bin maos --bin maos-wasm-runner && target/release/maos run examples/example-spirit-ts/manifest.toml --once 2>&1 | tee line2.log\nrc=${PIPESTATUS[0]}\nif grep -q '^SKIP ' line2.log; then rc=1; echo \"::error title=Epic-17 exit line 2 (${{ matrix.image }})::a SKIP line was printed\"; fi\nif [ \"$rc\" -ne 0 ]; then\n  echo \"::error title=Epic-17 exit line 2 (${{ matrix.image }})::rc=${rc} $(tail -n 3 line2.log | paste -sd' ' -)\"\nelse\n  echo \"::notice title=Epic-17 exit line 2 (${{ matrix.image }})::rc=0\"\nfi\nexit \"$rc\"\n"
    - uses: actions/upload-artifact@v4
      if: always()
      with:
        name: epic-17-exit-${{ matrix.image }}
        path: 'epic-17-exit-line1/

          t2.log

          line2.log

          '
        if-no-files-found: warn
  t3-smoke-busybox:

      fail-fast: false
      matrix:
        image:
        - ubuntu-24.04
        - ubuntu-26.04
    runs-on: ${{ matrix.image }}
    timeout-minutes: 30
    steps:
    - uses: actions/checkout@v5
    - uses: dtolnay/rust-toolchain@v1
      with:
        toolchain: stable
    - uses: Swatinem/rust-cache@v2
      with:
        key: t3-smoke-${{ matrix.image }}-${{ hashFiles('**/Cargo.lock') }}
    - uses: ./.github/actions/provision-t3-test-substrate
    - name: Run the live T3 boundary test
      shell: bash
      run: "set +e\ncargo test --locked -p maos-kernel-core --test sandbox_t3_spawn_linux --release -- --nocapture 2>&1 | tee t3-smoke.log\nrc=${PIPESTATUS[0]}\ngrep -E 'panicked at' t3-smoke.log | head -3 | while IFS= read -r line; do\n  echo \"::error title=T3 smoke (${{ matrix.image }})::${line}\"\ndone\nif grep -q '^SKIP ' t3-smoke.log; then rc=1; echo \"::error title=T3 smoke (${{ matrix.image }})::a SKIP line was printed\"; fi\necho \"::notice title=T3 smoke (${{ matrix.image }})::rc=${rc} $(grep '^test result' t3-smoke.log)\"\nexit \"$rc\"\n"

    strategy:
      fail-fast: false
      matrix:
        image:
        - ubuntu-24.04
        - ubuntu-26.04
    runs-on: ${{ matrix.image }}
    timeout-minutes: 30
    steps:
    - uses: actions/checkout@v5
    - uses: dtolnay/rust-toolchain@v1
      with:
        toolchain: stable
    - uses: Swatinem/rust-cache@v2
      with:
        key: t3-escape-corpus-${{ matrix.image }}-${{ hashFiles('**/Cargo.lock') }}
    - uses: ./.github/actions/provision-t3-test-substrate
    - name: Build the static network-escape probe
      run: RUSTFLAGS='-C target-feature=+crt-static' cargo build --locked --release --target x86_64-unknown-linux-gnu --manifest-path crates/maos-eval/test-fixtures/t3-network-escape-probe/Cargo.toml
    - name: Run the T3 escape corpus and the network executor
      shell: bash
      run: "set +e\ncargo test --locked -p maos-eval --test t3_escape_corpus --test t3_network_escape_executor --release -- --nocapture 2>&1 | tee t3-escape.log\nrc=${PIPESTATUS[0]}\nexecuted=$(grep -c '^T3-ESCAPE-EXECUTED ' t3-escape.log)\nlabelled=$(grep -c '^T3-ESCAPE-UNEXECUTED .* owner=20-3a-gate-honesty$' t3-escape.log)\ngrep '^T3-ESCAPE-EXECUTED ' t3-escape.log | while IFS= read -r line; do\n  echo \"::notice title=T3 escape corpus (${{ matrix.image }})::${line#T3-ESCAPE-EXECUTED }\"\ndone\ngrep '^T3-NETWORK-MATRIX ' t3-escape.log | while IFS= read -r line; do\n  echo \"::notice title=T3 escape corpus (${{ matrix.image }})::${line}\"\ndone\ngrep -E 'panicked at' t3-escape.log | head -3 | while IFS= read -r line; do\n  echo \"::error title=T3 escape corpus (${{ matrix.image }})::${line}\"\ndone\nif grep -q '^SKIP ' t3-escape.log; then rc=1; echo \"::error title=T3 escape corpus (${{ matrix.image }})::a SKIP line was printed\"; fi\nif [ \"$executed\" -ne 5 ] || [ \"$labelled\" -ne 20 ]; then\n  rc=1\n  echo \"::error title=T3 escape corpus (${{ matrix.image }})::executed=$executed (want 5) labelled=$labelled (want 20)\"\nfi\necho \"::notice title=T3 escape corpus (${{ matrix.image }})::rc=${rc} executed=${executed}/25 unexecuted=${labelled} owner=20-3a-gate-honesty\"\nexit \"$rc\"\n"
